void inference(double x1, double x2) {
  double u1=0;
  double u2=0;
  double Ux =0 ;
  for (int i=1; i<=configN; i++) {
    if (x1>=MF[i][1].a1 && x1<= MF[i][1].a4) {
      u1=MF[i][1].getu(x1);
      for (int j=1; j<=configN; j++) {
        if (x2>=MF[j][2].a1 && x2<= MF[j][2].a4) {
          u2=MF[j][2].getu(x2);
          Ux = Dmin(u1, u2);
          MF[rules_Table[j-1][i-1]][0].setfct(Ux);
        }
      }
    }
  }
}


double Defuzzification() {
  double sum_=0;
  double count_=0;
  for (int k=0; k<1000; k++) {
    sum_+=UoD[k]*((k-500)/5);
    count_+=UoD[k];
  }
  count_ = (count_ == 0) ? 1 : count_;
  arrayCopy(EmptyUoD, UoD);
  return sum_/count_;
}

double Dmin(double a, double b) {
  if (a < b)
    return a;
  else
    return b;
}

double Dmax(double a, double b) {
  if (a > b)
    return a;
  else
    return b;
}

class mbmfct {
  double a1;
  double a2;
  double a3;
  double a4;
  double a12;
  double a23;
  mbmfct() {
  }
  void inifct(double a, double b, double c, double d) {
    a1=a;
    a2=b;
    a3=c;
    a4=d;
  }

  int refineParameter(double A) {
    if (A >0)
      return int((float)A);
    else
      return -int(abs((float)A));
  }

  void setfct(double u) {
    double yy=0;
    a12=int((float) (u*(a2-a1)+a1));
    a23=int((float) (u*(a3-a4)+a4));
    a12 = refineParameter(a12);
    a23 = refineParameter(a23);
    int A1  = int((float)a1*5+500);
    int A12 = int((float)a12*5+500);
    int A23 = int((float)a23*5+500);
    int A4  = int((float)a4*5+500);
    ///////////////////////////////////////////////////////////////////////////////
    for (int i=A1; i<A12; i++) {
      double X = (i-500)/5;
      yy=((u)/(a12-a1))*(X-a1);
      UoD[i]= Dmax(yy, UoD[i]);
    }
    ///////////////////////////////////////////////////////////////////////////////
    for (int i=A12; i<A23; i++) {
      UoD[i]= Dmax(u, UoD[i]);
    }
    ////////////////////////////////////////////////////////////////////////////////
    for (int i=A23; i<A4; i++) {
      double X = (i-500)/5;
      yy=((-u)/(a4-a23))*(X-a4);
      UoD[i]= Dmax(yy, UoD[i]);
    }
  }
  ////////////////////////////////////////////////////////////////////////////////

  double getu(double x) {
    double returned_U=0;
    if (x>=a2 && x<=a3) {
      returned_U=1;
    } else if (x<a2) {
      returned_U=((x-a1)/(a2-a1));
    } else if (x>a3) {
      returned_U=((x-a4)/(a3-a4));
    }
    return returned_U;
  }
}
